```{r funciones}
plot.corr <- function(var){
  plot <- df |> select(where(is.numeric)) |>
  select(FLL:FVC, get(var)) |>
  ggcorr(hjust = 0.9, size = 2, layout.exp = 2, label = F, label_size = 2, label_round = 2, low = "darkred", high = "#0072B2", limits = c(-1, 1)) +
  ggtitle(paste0("Matriz de correlaciones ", var )) +
  guides(fill = guide_colourbar(title="Coef. correlación\nPearson"))
  return(plot)
}

# Tabla de correlación biserial por rangos (Wilcoxon) entre una variable
# dicotomizada (presencia/ausencia de defecto) y un conjunto de variables continuas.
tabla.biserial <- function(data, var.bin, vars.cont) {
  map_dfr(vars.cont, function(v) {
    x <- data[[v]]
    g <- data[[var.bin]]
    ok <- complete.cases(x, g)
    x <- x[ok]; g <- g[ok]
    n1 <- sum(g == levels(g)[1])
    n2 <- sum(g == levels(g)[2])
    if (n1 < 3 || n2 < 3 || length(unique(x)) < 2) {
      return(tibble(variable = v, n.no = n1, n.si = n2, r.biserial = NA_real_, p.valor = NA_real_))
    }
    test <- wilcox.test(x ~ g, exact = FALSE)
    r_rb <- 1 - 2 * unname(test$statistic) / (n1 * n2)
    tibble(variable = v, n.no = n1, n.si = n2, r.biserial = r_rb, p.valor = test$p.value)
  }) |> arrange(p.valor)
}

# Boxplots de un conjunto de variables continuas según una variable dicotomizada.
plot.boxplot.bin <- function(data, var.bin, vars.cont) {
  data |>
    select(all_of(c(var.bin, vars.cont))) |>
    pivot_longer(-all_of(var.bin), names_to = "variable", values_to = "valor") |>
    ggplot(aes(x = .data[[var.bin]], y = valor, fill = .data[[var.bin]])) +
    geom_boxplot() +
    facet_wrap(~variable, scales = "free_y") +
    scale_fill_manual(values = c("No" = color1, "Sí" = color2)) +
    labs(x = NULL, y = NULL) +
    theme(legend.position = "none")
}

# Correlación de Kendall entre la magnitud del defecto y las variables
# independientes, restringida a los individuos que presentan el defecto.
tabla.correlacion.magnitud <- function(data, var.dep, vars.ind) {
  sub <- data |> filter(.data[[var.dep]] > 0)
  map_dfr(vars.ind, function(v) {
    test <- suppressWarnings(cor.test(sub[[var.dep]], sub[[v]], method = "kendall"))
    tibble(variable = v, n = nrow(sub), tau = unname(test$estimate), p.valor = test$p.value)
  }) |> arrange(p.valor)
}
```

## Correlación con variables de baja variabilidad (HIP.D, HIP.I, PRE.D, PRE.I)

Las variables HIP.D, HIP.I, PRE.D y PRE.I apenas presentan variabilidad (entre el 73% y el 80% de los individuos tienen valor 0), lo que hace que los coeficientes de correlación habituales (Pearson, Spearman, Kendall) tengan muy poca potencia y sean poco fiables para estas variables. En su lugar se analizan en dos partes:

1. **Presencia del defecto**: se dicotomiza la variable (0 = sin defecto, >0 = con defecto) y se compara la distribución de cada variable independiente entre ambos grupos mediante el test de Wilcoxon-Mann-Whitney, con el coeficiente de correlación biserial por rangos como tamaño del efecto.
2. **Magnitud del defecto**: restringido a los individuos que presentan el defecto, se calcula la correlación de Kendall entre la magnitud del defecto y cada variable independiente.
